{-# OPTIONS --cubical-compatible --safe --no-sized-types --no-guardedness --level-universe #-}

module Agda.Builtin.UUID where

open import Agda.Builtin.Bool
open import Agda.Builtin.Maybe
open import Agda.Builtin.String

postulate UUID : Set
{-# BUILTIN UUID UUID #-}

primitive
  primUUIDEquality : UUID → UUID → Bool
  primShowUUID     : UUID → String
  primUUIDToString : UUID → String
  primStringToUUID : String → Maybe UUID

{-# COMPILE JS primUUIDEquality = function(x) { return function(y) { return x===y; }; } #-}
{-# COMPILE JS primShowUUID = function(x) { return "uuid\"" + x + "\""; } #-}
{-# COMPILE JS primUUIDToString = function(x) { return x; } #-}
{-# COMPILE JS primStringToUUID = function(x) {
  return /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/.test(x)
    ? z_jAgda_Agda_Builtin_Maybe["Maybe"]["just"](x)
    : z_jAgda_Agda_Builtin_Maybe["Maybe"]["nothing"];
  }
 #-}
